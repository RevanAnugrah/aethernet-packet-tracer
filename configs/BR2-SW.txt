! AetherNet - BR2-SW (Branch 2 multilayer switch)
!
enable
configure terminal
hostname BR2-SW
no ip domain-lookup
!
ip routing
!
vlan 81
 name DATA-BR2
vlan 82
 name VOICE-BR2
!
interface g1/0/1
 description To BR2
 no switchport
 ip address 10.0.0.54 255.255.255.252
 no shutdown
interface vlan 81
 ip address 10.1.81.1 255.255.255.0
 no shutdown
interface vlan 82
 ip address 10.1.82.1 255.255.255.0
 no shutdown
!
ip route 0.0.0.0 0.0.0.0 10.0.0.53
!
ip dhcp excluded-address 10.1.81.1 10.1.81.10
ip dhcp pool BR2-DATA
 network 10.1.81.0 255.255.255.0
 default-router 10.1.81.1
 dns-server 10.1.60.10
 domain-name aethernet.local
!
ip dhcp excluded-address 10.1.82.1 10.1.82.10
ip dhcp pool BR2-VOICE
 network 10.1.82.0 255.255.255.0
 default-router 10.1.82.1
 option 150 ip 10.0.0.38
!
interface range g1/0/2 - 3
 description Trunk to access switches
 switchport mode trunk
!
service timestamps log datetime msec
logging host 10.1.60.10
ip domain-name aethernet.local
ip ssh version 2
username <ADMIN_USER> privilege 15 secret <STRONG_PASSWORD>
line vty 0 15
 transport input ssh
 login local
!
end
