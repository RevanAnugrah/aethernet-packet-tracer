! AetherNet - D4 (Gedung 4 building switch)
!
enable
configure terminal
hostname D4
no ip domain-lookup
!
ip routing
!
vlan 41
 name DATA-G4
vlan 42
 name VOICE-G4
!
interface g1/1/3
 description To C1
 no switchport
 ip address 10.0.0.30 255.255.255.252
 no shutdown
interface g1/1/1
 description To C2
 no switchport
 ip address 10.0.0.34 255.255.255.252
 no shutdown
!
interface vlan 41
 ip address 10.1.41.1 255.255.255.0
 no shutdown
interface vlan 42
 ip address 10.1.42.1 255.255.255.0
 no shutdown
!
ip dhcp excluded-address 10.1.41.1 10.1.41.10
ip dhcp pool G4-DATA
 network 10.1.41.0 255.255.255.0
 default-router 10.1.41.1
 dns-server 10.1.60.10
 domain-name aethernet.local
!
ip dhcp excluded-address 10.1.42.1 10.1.42.10
ip dhcp pool G4-VOICE
 network 10.1.42.0 255.255.255.0
 default-router 10.1.42.1
 option 150 ip 10.0.0.38
!
router ospf 1
 router-id 10.255.0.14
 network 10.0.0.0 0.255.255.255 area 0
 passive-interface vlan 41
 passive-interface vlan 42
!
service timestamps log datetime msec
logging host 10.1.60.10
ip domain-name aethernet.local
ip ssh version 2
username <ADMIN_USER> privilege 15 secret <STRONG_PASSWORD>
line vty 0 15
 transport input ssh
 login local
!
end
